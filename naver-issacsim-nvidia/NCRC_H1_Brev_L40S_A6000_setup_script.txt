
#!/bin/bash
# Paste into Brev's Setup Script field.
# Shared H1/Go2 L40S / RTX A6000 base setup.
# It does not clone task source, start training, upload results,
# create a Release, or stop/delete the VM.
set -Eeuo pipefail

die() { echo "[ERROR] $*" >&2; exit 1; }
info() { echo "[INFO] $*"; }
[[ "$(uname -s)" == Linux ]] || die "Run in a Brev Ubuntu VM."
source /etc/os-release
[[ "${ID:-}" == ubuntu ]] || die "Ubuntu required: ${PRETTY_NAME:-unknown}"
case "${VERSION_ID:-}" in 22.04|24.04) ;; *) die "Ubuntu 22.04 or 24.04 required";; esac
command -v nvidia-smi >/dev/null || die "nvidia-smi is unavailable"
gpus="$(nvidia-smi --query-gpu=name --format=csv,noheader)"
echo "$gpus" | grep -qiE '(^|[[:space:]])(l40s|RTX A6000)([[:space:]]|$)' || die "L40S or RTX A6000 required; found: $gpus"

SIM_VER=5.1.0-rc.19
SIM_BUILD=5.1.0-rc.19+release.26219.9c81211b.gl
SIM_URL=https://downloads.isaacsim.nvidia.com/isaac-sim-standalone-5.1.0-linux-x86_64.zip
SIM_ROOT="${ISAAC_SIM_ROOT:-/isaac-sim}"
LAB_ROOT="${ISAACLAB_ROOT:-$HOME/IsaacLab}"
LAB_COMMIT=3e73d6dd79080fd7632488c061052a6edd52e230
CACHE="${HOME}/.cache/isaac-sim"
ZIP="$CACHE/isaac-sim-standalone-5.1.0-linux-x86_64.zip"

need_gib() { local p="$1" g="$2" a; mkdir -p "$p"; a="$(df -PB1 "$p" | awk 'NR==2{print $4}')"; (( a >= g * 1024 * 1024 * 1024 )) || die "$p needs ${g}GiB free"; }
sim_ok() { [[ -f "$SIM_ROOT/VERSION" && -x "$SIM_ROOT/python.sh" ]] && grep -q "^$SIM_VER" "$SIM_ROOT/VERSION"; }
if (( EUID == 0 )); then SUDO=(); else SUDO=(sudo); fi

info "Installing host tools"
"${SUDO[@]}" apt-get update
"${SUDO[@]}" env DEBIAN_FRONTEND=noninteractive apt-get install -y aria2 ca-certificates curl git git-lfs tmux unzip build-essential cmake libglu1-mesa
git lfs install --skip-repo
if ! command -v gh >/dev/null 2>&1; then
  key="$(mktemp)"
  curl -fsSL https://cli.github.com/packages/githubcli-archive-keyring.gpg -o "$key"
  "${SUDO[@]}" install -m 0644 "$key" /usr/share/keyrings/githubcli-archive-keyring.gpg
  rm -f "$key"
  echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/githubcli-archive-keyring.gpg] https://cli.github.com/packages stable main" | "${SUDO[@]}" tee /etc/apt/sources.list.d/github-cli.list >/dev/null
  "${SUDO[@]}" apt-get update
  "${SUDO[@]}" env DEBIAN_FRONTEND=noninteractive apt-get install -y gh
fi

mkdir -p "$CACHE"
download() {
  if command -v aria2c >/dev/null 2>&1; then
    aria2c --continue=true --max-connection-per-server=8 --split=8 --min-split-size=16M --file-allocation=none --dir="$CACHE" --out="$(basename "$ZIP")" "$SIM_URL"
  else
    curl --fail --location --retry 3 --continue-at - "$SIM_URL" -o "$ZIP"
  fi
}
if sim_ok; then
  info "Keeping Isaac Sim at $SIM_ROOT"
elif [[ -e "$SIM_ROOT" ]]; then
  die "$SIM_ROOT exists but is not $SIM_VER; refusing overwrite"
else
  need_gib "$CACHE" 25; need_gib "$(dirname "$SIM_ROOT")" 18
  [[ -s "$ZIP" ]] || download
  unzip -tq "$ZIP" || { download; unzip -tq "$ZIP" || die "Isaac Sim archive is corrupt"; }
  stage="$(mktemp -d "$CACHE/install.XXXXXX")"
  trap '[[ -n "${stage:-}" && -d "$stage" ]] && rm -rf "$stage"' EXIT
  unzip -q "$ZIP" -d "$stage"
  [[ -f "$stage/VERSION" && -x "$stage/python.sh" ]] || die "Unexpected Isaac Sim archive layout"
  grep -q "^$SIM_VER" "$stage/VERSION" || die "Isaac Sim version mismatch"
  "${SUDO[@]}" mv "$stage" "$SIM_ROOT"; stage=""
  (cd "$SIM_ROOT" && ./post_install.sh)
  sim_ok || die "Isaac Sim installation verification failed"
fi
grep -qF "$SIM_BUILD" "$SIM_ROOT/VERSION" || info "Expected RC19 build label not present; version matched: $(tr -d '\n' < "$SIM_ROOT/VERSION")"
"$SIM_ROOT/python.sh" -c 'import sys; assert sys.version_info[:3] == (3,11,13), sys.version'

if [[ -d "$LAB_ROOT/.git" ]]; then
  origin="$(git -C "$LAB_ROOT" remote get-url origin)"
  [[ "$origin" == https://github.com/isaac-sim/IsaacLab.git || "$origin" == https://github.com/isaac-sim/IsaacLab ]] || die "Unexpected Isaac Lab origin: $origin"
  [[ -z "$(git -C "$LAB_ROOT" status --porcelain)" ]] || die "Isaac Lab checkout is dirty"
  git -C "$LAB_ROOT" fetch origin "$LAB_COMMIT"
else
  [[ ! -e "$LAB_ROOT" ]] || die "$LAB_ROOT is not an Isaac Lab checkout"
  git clone --filter=blob:none https://github.com/isaac-sim/IsaacLab.git "$LAB_ROOT"
fi
git -C "$LAB_ROOT" checkout --detach "$LAB_COMMIT"
[[ "$(git -C "$LAB_ROOT" rev-parse HEAD)" == "$LAB_COMMIT" ]] || die "Isaac Lab commit mismatch"
ln -sfn "$SIM_ROOT" "$LAB_ROOT/_isaac_sim"
(cd "$LAB_ROOT" && ./isaaclab.sh -i rsl_rl)
ISAACLAB_PATH="$LAB_ROOT" "$SIM_ROOT/python.sh" - <<'PY'
import importlib.metadata, sys, torch, isaaclab, rsl_rl
assert sys.version_info[:3] == (3, 11, 13), sys.version
assert torch.cuda.is_available()
assert importlib.metadata.version("rsl-rl-lib") == "5.0.1"
print(torch.__version__, torch.cuda.get_device_name(0))
PY
"${SUDO[@]}" mkdir -p /workspace
"${SUDO[@]}" ln -sfn "$LAB_ROOT" /workspace/IsaacLab
[[ -x /workspace/IsaacLab/isaaclab.sh ]] || die "/workspace/IsaacLab link failed"
mkdir -p "$HOME/.config/issacsimz"
cat > "$HOME/.config/issacsimz/brev_rc19.env" <<EOF
export ISAAC_SIM_ROOT="$SIM_ROOT"
export ISAACLAB_ROOT="$LAB_ROOT"
export ISAACLAB_PATH="$LAB_ROOT"
EOF
info "Ready: Isaac Sim $SIM_VER; Isaac Lab $LAB_COMMIT"
cat <<'EOF'

Next, in an interactive VM terminal:
  1. Run the separately supplied scripts/ncrc_l40s_prepare.sh (GitHub login only).
  2. Clone the private NVIDIA source repository after login.
  3. Copy the checked issacsimz.pyz beside that checkout, then use doctor/train.
EOF
